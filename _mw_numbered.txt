  465:         private void ToggleCameraTileExpandCollapse(int tileIndex) {
  466:             if (IsSettingsDialogOpen()) {
  467:                 JsonLogStore.Information(
  468:                     "ExpandCollapseToggleSuppressed",
  469:                     "Expand/collapse toggle suppressed because the settings dialog is open.",
  470:                     InputDiagnosticsCategory,
  471:                     new Dictionary<string, object?> {
  472:                         ["tileIndex"] = tileIndex,
  473:                         ["expandedCameraIndex"] = _expandedCameraIndex
  474:                     });
  475:                 return;
  476:             }
  477: 
  478:             if (tileIndex < 0 || tileIndex >= _cameraTiles.Count) {
  479:                 JsonLogStore.Warning(
  480:                     "ExpandCollapseToggleRejected",
  481:                     "Expand/collapse toggle rejected because the camera tile index is out of range.",
  482:                     InputDiagnosticsCategory,
  483:                     new Dictionary<string, object?> {
  484:                         ["tileIndex"] = tileIndex,
  485:                         ["tileCount"] = _cameraTiles.Count
  486:                     });
  487:                 return;
  488:             }
  489: 
  490:             if (!IsCameraTileActive(tileIndex)) {
  491:                 JsonLogStore.Information(
  492:                     "ExpandCollapseToggleSuppressed",
  493:                     "Expand/collapse toggle suppressed because the camera tile is not active.",
  494:                     InputDiagnosticsCategory,
  495:                     new Dictionary<string, object?> {
  496:                         ["tileIndex"] = tileIndex,
  497:                         ["expandedCameraIndex"] = _expandedCameraIndex
  498:                     });
  499:                 return;
  500:             }
  501: 
  502:             var now = Environment.TickCount64;
  503:             if (_lastToggleTileIndex == tileIndex && now - _lastToggleTicks <= 150) {
  504:                 JsonLogStore.Information(
  505:                     "ExpandCollapseToggleSuppressed",
  506:                     "Duplicate expand/collapse toggle suppressed.",
  507:                     InputDiagnosticsCategory,
  508:                     new Dictionary<string, object?> {
  509:                         ["tileIndex"] = tileIndex,
  510:                         ["elapsedMs"] = now - _lastToggleTicks,
  511:                         ["expandedCameraIndex"] = _expandedCameraIndex
  512:                     });
  513:                 return;
  514:             }
  515: 
  516:             var previousExpandedIndex = _expandedCameraIndex;
  517:             _expandedCameraIndex = _expandedCameraIndex == tileIndex ? null : tileIndex;
  518:             _lastToggleTileIndex = tileIndex;
  519:             _lastToggleTicks = now;
  520: 
  521:             JsonLogStore.Information(
  522:                 "ExpandCollapseToggleApplied",
  523:                 "Expand/collapse toggle applied.",
  524:                 InputDiagnosticsCategory,
  525:                 new Dictionary<string, object?> {
  526:                     ["tileIndex"] = tileIndex,
  527:                     ["previousExpandedCameraIndex"] = previousExpandedIndex,
  528:                     ["newExpandedCameraIndex"] = _expandedCameraIndex,
  529:                     ["tileCount"] = _cameraTiles.Count
  530:                 });
  531: 
  532:             ApplyResponsiveCameraLayout();
  533:         }

  535:         private void InstallMouseHook() {
  536:             if (_mouseHookHandle != IntPtr.Zero && _lowLevelMouseHookHandle != IntPtr.Zero) {
  537:                 JsonLogStore.Information(
  538:                     "MouseHookInstallSkipped",
  539:                     "Mouse hook install skipped because hooks are already active.",
  540:                     InputDiagnosticsCategory,
  541:                     new Dictionary<string, object?> {
  542:                         ["hookHandle"] = _mouseHookHandle.ToInt64(),
  543:                         ["lowLevelHookHandle"] = _lowLevelMouseHookHandle.ToInt64(),
  544:                         ["threadId"] = GetCurrentThreadId()
  545:                     });
  546:                 return;
  547:             }
  548: 
  549:             var threadId = GetCurrentThreadId();
  550:             if (_mouseHookHandle == IntPtr.Zero) {
  551:                 _mouseHookProc = MouseHookCallback;
  552:                 _mouseHookHandle = SetWindowsHookEx(
  553:                     WhMouse,
  554:                     _mouseHookProc,
  555:                     IntPtr.Zero,
  556:                     threadId);
  557:             }
  558: 
  559:             var mouseHookError = Marshal.GetLastWin32Error();
  560:             if (_lowLevelMouseHookHandle == IntPtr.Zero) {
  561:                 _lowLevelMouseHookProc = LowLevelMouseHookCallback;
  562:                 _lowLevelMouseHookHandle = SetWindowsHookEx(
  563:                     WhMouseLl,
  564:                     _lowLevelMouseHookProc,
  565:                     IntPtr.Zero,
  566:                     0);
  567:             }
  568: 
  569:             var lowLevelHookError = Marshal.GetLastWin32Error();
  570:             JsonLogStore.Information(
  571:                 _mouseHookHandle == IntPtr.Zero && _lowLevelMouseHookHandle == IntPtr.Zero
  572:                     ? "MouseHookInstallFailed"
  573:                     : "MouseHookInstalled",
  574:                 "Mouse hook install attempted.",
  575:                 InputDiagnosticsCategory,
  576:                 new Dictionary<string, object?> {
  577:                     ["hookHandle"] = _mouseHookHandle.ToInt64(),
  578:                     ["lowLevelHookHandle"] = _lowLevelMouseHookHandle.ToInt64(),
  579:                     ["threadId"] = threadId,
  580:                     ["mouseHookLastWin32Error"] = mouseHookError,
  581:                     ["lowLevelHookLastWin32Error"] = lowLevelHookError
  582:                 });
  583:         }

  585:         private void UninstallMouseHook() {
  586:             var hookHandle = _mouseHookHandle;
  587:             var lowLevelHookHandle = _lowLevelMouseHookHandle;
  588:             if (_mouseHookHandle != IntPtr.Zero) {
  589:                 UnhookWindowsHookEx(_mouseHookHandle);
  590:                 _mouseHookHandle = IntPtr.Zero;
  591:                 _mouseHookProc = null;
  592:             }
  593: 
  594:             if (_lowLevelMouseHookHandle != IntPtr.Zero) {

  630:         private void HandleThreadMouseDown(int screenX, int screenY) {
  631:             if (IsSettingsDialogOpen()) {
  632:                 ResetDoubleClickTracking();
  633:                 return;
  634:             }
  635: 
  636:             var tileIndex = GetCameraTileIndexAtScreenPoint(screenX, screenY);
  637:             if (!tileIndex.HasValue) {
  638:                 LogUnmatchedMouseDown(screenX, screenY);
  639:                 ResetDoubleClickTracking();
  640:                 return;
  641:             }
  642:             if (!IsCameraTileActive(tileIndex.Value)) {
  643:                 ResetDoubleClickTracking();
  644:                 return;
  645:             }
  646: 
  647:             var now = Environment.TickCount64;
  648:             var isDoubleClick = _lastDoubleClickTileIndex == tileIndex.Value &&
  649:                                 _lastDoubleClickTicks > 0 &&
  650:                                 now - _lastDoubleClickTicks <= GetDoubleClickTime() &&
  651:                                 Math.Abs(screenX - _lastDoubleClickX) <= GetSystemMetrics(SmCxDoubleClk) &&
  652:                                 Math.Abs(screenY - _lastDoubleClickY) <= GetSystemMetrics(SmCyDoubleClk);
  653: 
  654:             JsonLogStore.Information(
  655:                 "MouseDownMatchedCameraCard",
  656:                 "Mouse down matched a camera card.",
  657:                 InputDiagnosticsCategory,
  658:                 new Dictionary<string, object?> {
  659:                     ["tileIndex"] = tileIndex.Value,
  660:                     ["screenX"] = screenX,
  661:                     ["screenY"] = screenY,
  662:                     ["previousTileIndex"] = _lastDoubleClickTileIndex,
  663:                     ["elapsedMs"] = _lastDoubleClickTicks > 0 ? now - _lastDoubleClickTicks : null,
  664:                     ["doubleClickTimeMs"] = GetDoubleClickTime(),
  665:                     ["doubleClickWidth"] = GetSystemMetrics(SmCxDoubleClk),
  666:                     ["doubleClickHeight"] = GetSystemMetrics(SmCyDoubleClk),
  667:                     ["isDoubleClick"] = isDoubleClick,
  668:                     ["expandedCameraIndex"] = _expandedCameraIndex
  669:                 });
  670: 
  671:             if (isDoubleClick) {
  672:                 ResetDoubleClickTracking();
  673:                 LogDoubleClickToggleAttempt("MouseHook", tileIndex.Value, screenX, screenY);
  674:                 ToggleCameraTileExpandCollapse(tileIndex.Value);
  675:                 return;
  676:             }
  677: 
  678:             _lastDoubleClickTileIndex = tileIndex.Value;
  679:             _lastDoubleClickTicks = now;
  680:             _lastDoubleClickX = screenX;
  681:             _lastDoubleClickY = screenY;
  682:         }

  684:         private int? GetCameraTileIndexAtScreenPoint(int screenX, int screenY) {
  685:             var screenPoint = new Point(screenX, screenY);
  686:             for (var i = 0; i < _cameraTiles.Count; i++) {
  687:                 var card = _cameraTiles[i].Card;
  688:                 if (!card.IsVisible || card.ActualWidth <= 0 || card.ActualHeight <= 0) {
  689:                     continue;
  690:                 }
  691: 
  692:                 var cardPoint = card.PointFromScreen(screenPoint);
  693:                 if (cardPoint.X >= 0 &&
  694:                     cardPoint.Y >= 0 &&
  695:                     cardPoint.X <= card.ActualWidth &&
  696:                     cardPoint.Y <= card.ActualHeight) {
  697:                     return i;
  698:                 }
  699:             }
  700: 
  701:             return null;
  702:         }

  890:         private async Task StartLocalCameraSearchAsync() {
  891:             if (_isClosing || _isScanning) {
  892:                 return;
  893:             }
  894: 
  895:             JsonLogStore.Information(
  896:                 eventName: "camera_search_requested",
  897:                 message: "User requested a local network camera search.",
  898:                 category: "camera_search",
  899:                 data: new Dictionary<string, object?> {
  900:                     ["source"] = "main_window",
  901:                     ["isClosing"] = _isClosing,
  902:                     ["isScanning"] = _isScanning
  903:                 });
  904: 
  905:             while (!_isClosing) {
  906:                 _isScanning = true;
  907:                 _scanCancellation = new CancellationTokenSource();
  908:                 var progress = new Progress<TapoCameraScanActivity>(activity => {
  909:                     if (_isClosing || !_isScanning) {
  910:                         return;
  911:                     }
  912: 
  913:                     StreamingStatusText.Text = activity.StatusMessage;
  914:                 });
  915:                 var preferredMethod = ResolvePreferredDetectionMethod();
  916:                 SetScanningState(preferredMethod);
  917: 
  918:                 try {
  919:                     var scanResult = await TapoCameraScanner.ScanLocalNetworkForTapoCamerasWithDiagnosticsAsync(
  920:                         preferredFirstMethod: preferredMethod,
  921:                         progress: progress,
  922:                         cancellationToken: _scanCancellation.Token);
  923: 
  924:                     if (_isClosing) {
  925:                         return;
  926:                     }
  927: 
  928:                     if (scanResult.Detections.Count > 0) {
  929:                         PersistSuccessfulDetectionMethod(scanResult.SuccessfulMethod);
  930:                         JsonLogStore.Information(
  931:                             eventName: "camera_search_detections_available",
  932:                             message: "Local network scan found one or more likely Tapo cameras.",
  933:                             category: "camera_search",
  934:                             data: new Dictionary<string, object?> {
  935:                                 ["detectionCount"] = scanResult.Detections.Count,
  936:                                 ["detectedIps"] = scanResult.Detections.Select(d => d.IpAddress.ToString()).ToArray(),
  937:                                 ["successfulMethod"] = scanResult.SuccessfulMethod?.ToString()
  938:                             });
  939:                         ShowDetections(scanResult.Detections, scanResult.SuccessfulMethod);
  940:                         return;
  941:                     }
  942: 
  943:                     JsonLogStore.Warning(
  944:                         eventName: "camera_search_no_detections",
  945:                         message: "Local network scan completed without any likely Tapo cameras.",
  946:                         category: "camera_search",
  947:                         data: new Dictionary<string, object?> {
  948:                             ["attemptedMethods"] = scanResult.AttemptedMethods.Select(a => a.Method.ToString()).ToArray()
  949:                         });
  950:                     ShowNoDetections(BuildNoDetectionsMessage(scanResult.AttemptedMethods));
  951:                 }
  952:                 catch (OperationCanceledException) when (_isClosing) {
  953:                     return;
  954:                 }
  955:                 catch (OperationCanceledException) {
  956:                     if (!_isClosing) {
  957:                         JsonLogStore.Warning(
  958:                             eventName: "camera_search_operation_canceled",
  959:                             message: "Local network search was canceled.",
  960:                             category: "camera_search");
  961:                         ShowNoDetections("Scan canceled.");
  962:                     }
  963:                 }
  964:                 catch (Exception ex) {
  965:                     if (!_isClosing) {
  966:                         JsonLogStore.Error(
  967:                             eventName: "camera_search_operation_failed",
  968:                             message: "Local network search failed in the main window.",
  969:                             category: "camera_search",
  970:                             exception: ex);
  971:                         ShowNoDetections($"Scan failed: {ex.Message}");
  972:                     }
  973:                 }
  974:                 finally {
  975:                     _scanCancellation?.Dispose();
  976:                     _scanCancellation = null;
  977:                     _isScanning = false;
  978:                 }
  979: 
  980:                 if (_isClosing) {
  981:                     return;
  982:                 }
  983: 
  984:                 var retry = MessageBox.Show(
  985:                     this,
  986:                     "No local camera was found. Retry search?",
  987:                     "LocalCam",
  988:                     MessageBoxButton.YesNo,
  989:                     MessageBoxImage.Question);
  990: 
  991:                 if (retry != MessageBoxResult.Yes) {
  992:                     return;
  993:                 }
  994:             }
  995:         }

 1091:         private void StartAllStreams() {
 1092:             if (_libVlc is null) {
 1093:                 JsonLogStore.Warning(
 1094:                     eventName: "camera_connect_blocked",
 1095:                     message: "RTSP stream start was blocked because the video engine is unavailable.",
 1096:                     category: "camera_connect");
 1097:                 StreamingStatusText.Text = "Cannot start streams: video engine is unavailable.";
 1098:                 return;
 1099:             }
 1100: 
 1101:             if (_detections.Count == 0) {
 1102:                 JsonLogStore.Warning(
 1103:                     eventName: "camera_connect_blocked",
 1104:                     message: "RTSP stream start was blocked because no cameras were detected.",
 1105:                     category: "camera_connect");
 1106:                 StreamingStatusText.Text = "Cannot start streams: no cameras were detected.";
 1107:                 return;
 1108:             }
 1109: 
 1110:             var username = _settings.RtspUsername.Trim();
 1111:             var password = _settings.RtspPassword;
 1112:             var streamPath = NormalizeStreamPath(_settings.StreamPath);
 1113: 
 1114:             if (!HasCompleteStreamingSettings(username, password)) {
 1115:                 JsonLogStore.Warning(
 1116:                     eventName: "camera_connect_blocked",
 1117:                     message: "RTSP stream start was blocked because credentials are incomplete.",
 1118:                     category: "camera_connect",
 1119:                     data: new Dictionary<string, object?> {
 1120:                         ["cameraCount"] = _detections.Count,
 1121:                         ["streamPath"] = streamPath
 1122:                     });
 1123:                 StreamingStatusText.Text = BuildMissingSettingsPrompt();
 1124:                 return;
 1125:             }
 1126: 
 1127:             JsonLogStore.Information(
 1128:                 eventName: "camera_connect_requested",
 1129:                 message: "Starting RTSP connections for detected cameras.",
 1130:                 category: "camera_connect",
 1131:                 data: new Dictionary<string, object?> {
 1132:                     ["cameraCount"] = _detections.Count,
 1133:                     ["streamPath"] = streamPath,
 1134:                     ["hasUsername"] = !string.IsNullOrWhiteSpace(username)
 1135:                 });
 1136: 
 1137:             _isStartingStreams = true;
 1138:             UpdateActionButtons();
 1139: 
 1140:             var startedCount = 0;
 1141:             var failedEndpoints = new List<string>();
 1142: 
 1143:             EnsureCameraTileCount(_detections.Count);
 1144:             for (var i = 0; i < _cameraTiles.Count && i < _detections.Count; i++) {
 1145:                 if (IsStreamRunning(i)) {
 1146:                     continue;
 1147:                 }
 1148: 
 1149:                 var ipAddress = _detections[i].IpAddress.ToString();
 1150:                 var streamUrl = BuildRtspUrl(ipAddress, username, password, streamPath);
 1151:                 var mediaPlayer = _cameraTiles[i].MediaPlayer;
 1152:                 if (mediaPlayer is null) {
 1153:                     failedEndpoints.Add(ipAddress);
 1154:                     SetVideoSurfaceActive(i, isActive: false);
 1155:                     continue;
 1156:                 }
 1157: 
 1158:                 try {
 1159:                     JsonLogStore.Information(
 1160:                         eventName: "camera_connect_attempt",
 1161:                         message: "Attempting to start an RTSP stream.",
 1162:                         category: "camera_connect",
 1163:                         data: new Dictionary<string, object?> {
 1164:                             ["cameraIndex"] = i + 1,
 1165:                             ["ipAddress"] = ipAddress,
 1166:                             ["streamPath"] = streamPath
 1167:                         });
 1168: 
 1169:                     using var media = new Media(_libVlc, streamUrl, FromType.FromLocation);

 1492:         private static (int Columns, int Rows, double CardWidth, double CardHeight) CalculateVideoDrivenGrid(
 1493:             int count,
 1494:             double panelWidth,
 1495:             double panelHeight) {
 1496:             const double videoAspect = 16d / 9d;
 1497:             const double gridGap = 6d;
 1498:             const double cardChromeHorizontal = 8d;
 1499:             const double cardChromeVertical = 8d;
 1500: 
 1501:             var bestColumns = 1;
 1502:             var bestRows = count;
 1503:             var bestCardWidth = panelWidth;
 1504:             var bestCardHeight = panelHeight / Math.Max(1, count);
 1505:             var bestVideoArea = double.NegativeInfinity;
 1506: 
 1507:             for (var columns = 1; columns <= count; columns++) {
 1508:                 var rows = (int)Math.Ceiling(count / (double)columns);
 1509:                 var totalGapWidth = Math.Max(0, columns - 1) * (2d * gridGap);
 1510:                 var totalGapHeight = Math.Max(0, rows - 1) * (2d * gridGap);
 1511:                 var slotWidth = Math.Max(1d, (panelWidth - totalGapWidth) / columns);
 1512:                 var slotHeight = Math.Max(1d, (panelHeight - totalGapHeight) / rows);
 1513:                 var maxVideoWidth = Math.Max(1d, slotWidth - cardChromeHorizontal);
 1514:                 var maxVideoHeight = Math.Max(1d, slotHeight - cardChromeVertical);
 1515: 
 1516:                 var videoWidth = Math.Min(maxVideoWidth, maxVideoHeight * videoAspect);
 1517:                 var videoHeight = videoWidth / videoAspect;
 1518: 
 1519:                 if (videoHeight > maxVideoHeight) {
 1520:                     videoHeight = maxVideoHeight;
 1521:                     videoWidth = videoHeight * videoAspect;
 1522:                 }
 1523: 
 1524:                 var cardWidth = Math.Max(1d, videoWidth + cardChromeHorizontal);
 1525:                 var cardHeight = Math.Max(1d, videoHeight + cardChromeVertical);
 1526:                 var videoArea = videoWidth * videoHeight;
 1527: 
 1528:                 if (videoArea > bestVideoArea) {
 1529:                     bestVideoArea = videoArea;
 1530:                     bestColumns = columns;
 1531:                     bestRows = rows;
 1532:                     bestCardWidth = cardWidth;
 1533:                     bestCardHeight = cardHeight;
 1534:                 }
 1535:             }
 1536: 
 1537:             return (bestColumns, bestRows, bestCardWidth, bestCardHeight);
 1538:         }
